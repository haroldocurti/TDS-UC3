2. Fundamentos de DQL: Seleção, Filtros e Ordenação

SQL • SELECT, WHERE, ORDER BY e LIMIT MySQL 
-- 1. Seleção de colunas específicas
SELECT nome, email, cidade FROM clientes;

-- 2. Filtros combinados com operadores relacionais e lógicos
SELECT nome, preco, estoque 
FROM produtos 
WHERE categoria = 'Eletrônicos' AND preco > 100.00;

-- 3. Filtros por intervalo (BETWEEN) e listas (IN)
SELECT nome, preco FROM produtos WHERE preco BETWEEN 50.00 AND 100.00;
SELECT * FROM clientes WHERE estado IN ('SP', 'RJ', 'MG');

-- 4. Busca por padrões de texto com LIKE (% representa múltiplos caracteres)
SELECT nome, email FROM clientes WHERE nome LIKE 'M%' AND email LIKE '%@email.com';

-- 5. Ordenação e paginação (LIMIT)
SELECT nome, preco, estoque FROM produtos ORDER BY preco DESC LIMIT 5;

3. Funções de Agregação, Agrupamentos e HAVING

Funções agregadas processam conjuntos de linhas e retornam métricas de resumo:

SQL • GROUP BY e Métricas Agregadas MySQL 
SELECT 
    categoria,
    COUNT(*) AS total_produtos,
    AVG(preco) AS media_preco,
    MIN(preco) AS menor_preco,
    MAX(preco) AS maior_preco
FROM produtos
GROUP BY categoria
HAVING media_preco > 150.00
ORDER BY media_preco DESC;
Cláusula Momento de Execução Uso Recomendado
WHERE Filtra linhas individuais antes do agrupamento. WHERE status = 'Processando'
HAVING Filtra os grupos resultantes após o GROUP BY. HAVING SUM(subtotal) > 300.00

4. Junções Relacionais: INNER JOIN, LEFT JOIN e Cadeias

SQL • INNER JOIN (Cruzamento Exato) MySQL 
SELECT 
    clientes.nome AS nome_do_cliente,
    pedidos.id AS numero_do_pedido,
    pedidos.data_pedido
FROM clientes
INNER JOIN pedidos ON clientes.id = pedidos.cliente_id;
SQL • LEFT JOIN e Detecção de Inativos (IS NULL) MySQL 
-- Identifica clientes cadastrados que nunca realizaram nenhum pedido
SELECT 
    clientes.nome,
    clientes.email
FROM clientes
LEFT JOIN pedidos ON clientes.id = pedidos.cliente_id
WHERE pedidos.id IS NULL;
SQL • Junção Multi-Tabelas em Cadeia (4 Tabelas Conectadas) MySQL 
SELECT 
    clientes.nome AS cliente,
    pedidos.id AS pedido_id,
    produtos.nome AS produto,
    itens_pedido.quantidade,
    itens_pedido.preco_unitario,
    (itens_pedido.quantidade * itens_pedido.preco_unitario) AS subtotal
FROM clientes
INNER JOIN pedidos ON clientes.id = pedidos.cliente_id
INNER JOIN itens_pedido ON pedidos.id = itens_pedido.pedido_id
INNER JOIN produtos ON itens_pedido.produto_id = produtos.id;
